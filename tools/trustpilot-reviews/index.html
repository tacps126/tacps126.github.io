Python · pip install apify-client
from apify_client import ApifyClient

client = ApifyClient("YOUR_APIFY_TOKEN")
run = client.actor("tacps126/trustpilot-reviews").call(run_input={
    "companies": ["apify.com", "https://www.trustpilot.com/review/github.com"],
    "maxReviews": 200,
    "stars": 5,
    "language": "en",
})
for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
JavaScript · npm i apify-client
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: 'YOUR_APIFY_TOKEN' });
const run = await client.actor('tacps126/trustpilot-reviews').call({
  "companies": [
    "apify.com",
    "https://www.trustpilot.com/review/github.com"
  ],
  "maxReviews": 200,
  "stars": 5,
  "language": "en"
});
const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
